\begin{frame}{TCP frente a UDP}
  \begin{columns}[T]
    \begin{column}{0.47\textwidth}
      \begin{tarjeta}[equal height group=tcpudp]{bcmTCP}{TCP}
        \begin{itemize}
          \item Conexión previa y entrega en orden
          \item Costo: handshake, ACK, más latencia
          \item Ej.: navegador, SGA, correo
        \end{itemize}
      \end{tarjeta}
    \end{column}
    \begin{column}{0.47\textwidth}
      \begin{tarjeta}[equal height group=tcpudp]{bcmUDP}{UDP}
        \begin{itemize}
          \item Sin conexión ni reenvío
          \item Gana: menos espera, cabecera de 8 bytes
          \item Ej.: Meet, juegos en línea, DNS
        \end{itemize}
      \end{tarjeta}
    \end{column}
  \end{columns}
  \vspace{0.35cm}
  {\small\color{bcmGris}Fuente: \cite[pp.~7--8]{barghash2022}\par}
\end{frame}
